Docker Compose Quickstart

Launch a complete private OCI registry + ZotDeck web dashboard in seconds:

version: "3.8"

services:
  zot:
    image: ghcr.io/project-zot/zot-linux-amd64:latest
    ports:
      - "5000:5000"
    volumes:
      - ./data:/var/lib/registry

  zotdeck:
    image: bhaveshp36/zotdeck:latest
    ports:
      - "8089:8000"
    environment:
      - ZOT_ENDPOINT=http://zot:5000
    depends_on:
      - zot
